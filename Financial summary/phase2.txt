from dataclasses import dataclass
from datetime import date


# ============================================================
# Transaction Data Structure
# ============================================================

@dataclass
class Transaction:
    type: str
    amount: float
    category: str
    description: str
    date: str


transactions = []


# ============================================================
# Helper Function - Get Valid Amount
# ============================================================

def get_amount():
    while True:
        try:
            amount = float(input("Enter amount: "))

            if amount <= 0:
                print("Amount must be greater than 0.")
                continue

            return amount

        except ValueError:
            print("Invalid amount. Please enter a number.")


# ============================================================
# Add Income
# ============================================================

def add_income():

    amount = get_amount()

    category = input("Enter category: ").strip()
    description = input("Enter description: ").strip()

    transaction = Transaction(
        type="income",
        amount=amount,
        category=category,
        description=description,
        date=str(date.today())
    )

    transactions.append(transaction)

    print("\nIncome added successfully!")


# ============================================================
# Add Expense
# ============================================================

def add_expense():

    amount = get_amount()

    category = input("Enter category: ").strip()
    description = input("Enter description: ").strip()

    transaction = Transaction(
        type="expense",
        amount=amount,
        category=category,
        description=description,
        date=str(date.today())
    )

    transactions.append(transaction)

    print("\nExpense added successfully!")


# ============================================================
# Display Transactions
# ============================================================

def display_transactions(transaction_list):

    if len(transaction_list) == 0:
        print("\nNo transactions found.")
        return

    print("\n" + "=" * 90)
    print("TRANSACTIONS")
    print("=" * 90)

    print(
        f"{'ID':<5}"
        f"{'DATE':<15}"
        f"{'TYPE':<12}"
        f"{'AMOUNT':<15}"
        f"{'CATEGORY':<15}"
        f"{'DESCRIPTION'}"
    )

    print("-" * 90)

    for index, transaction in enumerate(transaction_list, start=1):

        print(
            f"{index:<5}"
            f"{transaction.date:<15}"
            f"{transaction.type.upper():<12}"
            f"৳{transaction.amount:<14.2f}"
            f"{transaction.category:<15}"
            f"{transaction.description}"
        )

    print("=" * 90)


# ============================================================
# View Transactions
# ============================================================

def view_transactions():

    if len(transactions) == 0:
        print("\nNo transactions yet.")
        return

    print("\nHow do you want to sort?")

    print("1. Newest first")
    print("2. Oldest first")
    print("3. Highest amount")
    print("4. Lowest amount")

    choice = input("Enter choice: ")

    if choice == "1":

        sorted_transactions = sorted(
            transactions,
            key=lambda transaction: transaction.date,
            reverse=True
        )

    elif choice == "2":

        sorted_transactions = sorted(
            transactions,
            key=lambda transaction: transaction.date
        )

    elif choice == "3":

        sorted_transactions = sorted(
            transactions,
            key=lambda transaction: transaction.amount,
            reverse=True
        )

    elif choice == "4":

        sorted_transactions = sorted(
            transactions,
            key=lambda transaction: transaction.amount
        )

    else:

        print("Invalid choice.")
        return

    display_transactions(sorted_transactions)


# ============================================================
# Search Transactions
# ============================================================

def search_transactions():

    if len(transactions) == 0:
        print("\nNo transactions yet.")
        return

    category = input(
        "\nEnter category to search: "
    ).strip().lower()

    results = [
        transaction
        for transaction in transactions
        if transaction.category.lower() == category
    ]

    if len(results) == 0:
        print("\nNo transactions found for this category.")
        return

    print(f"\nFound {len(results)} transaction(s).")

    display_transactions(results)


# ============================================================
# Category Summary
# ============================================================

def category_summary():

    summary = {}

    for transaction in transactions:

        if transaction.type == "expense":

            category = transaction.category

            if category not in summary:
                summary[category] = 0

            summary[category] += transaction.amount

    if len(summary) == 0:
        print("\nNo expenses found.")
        return

    print("\n========== EXPENSE BY CATEGORY ==========")

    for category, amount in sorted(
        summary.items(),
        key=lambda item: item[1],
        reverse=True
    ):

        print(f"{category:<20} : ৳{amount:.2f}")


# ============================================================
# Show Balance
# ============================================================

def show_balance():

    total_income = sum(
        transaction.amount
        for transaction in transactions
        if transaction.type == "income"
    )

    total_expense = sum(
        transaction.amount
        for transaction in transactions
        if transaction.type == "expense"
    )

    balance = total_income - total_expense

    print("\n========== BALANCE ==========")

    print(f"Total Income : ৳{total_income:.2f}")
    print(f"Total Expense: ৳{total_expense:.2f}")
    print("-----------------------------")
    print(f"Balance      : ৳{balance:.2f}")


# ============================================================

    print("\n")
    print("╔══════════════════════════════════════╗")
    print("║       PERSONAL FINANCE MANAGER       ║")

    print("\n1. Add Income")
    print("2. Add Expense")
    print("3. View Transactions")
    print("4. Search Transactions")
    print("5. Category Summary")
    print("6. Show Balance")
    print("7. Exit")


# ============================================================
# Main Program
# ============================================================

while True:

    show_menu()

    choice = input("\nEnter your choice: ").strip()

    if choice == "1":
        add_income()

    elif choice == "2":
        add_expense()

    elif choice == "3":
        view_transactions()

    elif choice == "4":
        search_transactions()

    elif choice == "5":
        category_summary()

    elif choice == "6":
        show_balance()

    elif choice == "7":

        print("\nThank you for using Personal Finance Manager!")
        break

    else:
        print("\nInvalid choice. Please try again.")
